First Install few dependent packages and Docker Application.

$ apt update && apt install -y apt-transport-https ca-certificates curl software-properties-common docker.io -y

# After Docker Installation, we can start docker service using below command:
$ systemctl enable docker.service

# Verify the Docker Installation by running below command:
$ docker --version

## Now Proceed with K8s Installations ## First add gpg-key and then update repo details by cat command.

$ curl -s https://packages.cloud.google.com/apt/doc/apt-key.gpg | apt-key add -

$ cat <<EOF >/etc/apt/sources.list.d/kubernetes.list
deb https://apt.kubernetes.io/ kubernetes-xenial main
EOF

# Install the Kubernetes packages by below command:	
$ apt update && apt install -y kubelet kubeadm kubectl

# Now, load the daemon and restart the service by below command:
$ systemctl daemon-reload && systemctl restart kubelet

========================RUN Below Commands only on Master Node======================
# To Initialize kubernetes on Master Node (only on Master Node):

$ kubeadm init
